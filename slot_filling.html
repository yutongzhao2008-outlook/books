① 什么是槽位填充

槽位填充是任务型对话（Task-Oriented Dialogue）的核心环节。系统先识别用户意图（Intent），再把句子中的关键信息抽取为槽位（Slot），最后检查哪些必填槽位仍然为空，并向用户追问。

示例：用户说 “帮我订明天从北京到上海的高铁票”

意图 = book_train from = 北京 to = 上海 date = 2026-10-20 seat = 缺失 → 追问“一等座还是二等座？”

传统方案 vs. LLM 方案

维度规则 / 序列标注（BIO + CRF）LLM 抽取（Prompt + JSON）
需要标注数据需要大量token级标注几乎零样本，Schema 即可
新增槽位重新标注、重新训练修改 Schema 与 Prompt
口语/省略/指代较弱较强（可结合对话历史）
输出稳定性确定需要约束格式、校验与重试

⑤ 参考实现（Python）

  

    基于 OpenAI 兼容接口，可替换为任意支持 JSON 输出的模型服务
    
  


import json
import re
from openai import OpenAI

client = OpenAI()            # 任何兼容 OpenAI 接口的模型服务都可以
TODAY = "2026-10-19"

SCHEMA = {
    "intent": "book_train",
    "slots": {
        "from": {"desc": "出发城市", "required": True, "ask": "请问从哪个城市出发？"},
        "to":   {"desc": "到达城市", "required": True, "ask": "请问要到哪个城市？"},
        "date": {"desc": "出发日期，格式 YYYY-MM-DD", "required": True,
                 "ask": "请问哪天出发？", "pattern": r"^\d{4}-\d{2}-\d{2}$"},
        "seat": {"desc": "座位等级：一等座/二等座/商务座", "required": True,
                 "ask": "请问您要一等座、二等座还是商务座？",
                 "enum": ["一等座", "二等座", "商务座"]},
    },
}

def build_messages(utterance, state):
    """步骤②：构造 Prompt（指令 + Schema + 已知状态 + 用户输入）"""
    slot_lines = "\n".join(
        f"- {name}（{cfg['desc']}）" for name, cfg in SCHEMA["slots"].items()
    )
    system = (
        f"你是槽位抽取器。当前日期：{TODAY}。意图：{SCHEMA['intent']}\n"
        f"需要抽取的槽位：\n{slot_lines}\n"
        "规则：只输出 JSON 对象，键为槽位名；用户未提及的槽位填 null；"
        "与已知状态冲突时以最新输入为准；不要编造。"
    )
    user = f"已知状态：{json.dumps(state, ensure_ascii=False)}\n用户输入：{utterance}"
    return [{"role": "system", "content": system},
            {"role": "user", "content": user}]

def call_llm(utterance, state, retries=1):
    """步骤③④：调用模型，强制 JSON 输出，解析失败则重试"""
    for _ in range(retries + 1):
        resp = client.chat.completions.create(
            model="your-model",
            messages=build_messages(utterance, state),
            temperature=0,                                # 抽取任务要确定性
            response_format={"type": "json_object"},      # 约束输出为 JSON
        )
        try:
            return json.loads(resp.choices[0].message.content)
        except json.JSONDecodeError:
            continue
    return {}

def merge(state, new_slots):
    """步骤⑥：合并状态，null 不覆盖已有值"""
    for key, value in new_slots.items():
        if value is not None:
            state[key] = value
    return state

def validate(state):
    """步骤⑤：代码侧校验，不完全相信模型输出"""
    errors = {}
    for name, cfg in SCHEMA["slots"].items():
        value = state.get(name)
        if value in (None, ""):
            if cfg["required"]:
                errors[name] = "缺失"
        elif "enum" in cfg and value not in cfg["enum"]:
            errors[name] = "取值不合法"
        elif "pattern" in cfg and not re.match(cfg["pattern"], str(value)):
            errors[name] = "格式不正确"
    return errors

def run_turn(utterance, state):
    """一轮对话：抽取 → 合并 → 校验 → 追问或执行"""
    state = merge(state, call_llm(utterance, state))
    errors = validate(state)
    if not errors:
        return state, "已为您提交订单。"
    first_missing = next(iter(errors))
    return state, SCHEMA["slots"][first_missing]["ask"]

if __name__ == "__main__":
    state = {}
    for text in ["帮我订明天从北京到上海的高铁票", "二等座"]:
        state, reply = run_turn(text, state)
        print("用户:", text)
        print("状态:", state)
        print("助手:", reply)
